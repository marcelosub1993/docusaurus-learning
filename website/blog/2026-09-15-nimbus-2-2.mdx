    </TabItem>

    <TabItem value="macos" label="macOS">

        ```bash title="Terminal" showLineNumbers

        ```

    </TabItem>

    <TabItem value="linux" label="Linux" >

        ```bash title="Terminal" showLineNumbers
            sudo apt upgrade Nimbus -y
        ```

    </TabItem>
